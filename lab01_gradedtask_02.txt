2. Verbatim AI Suggestion

try:
    start = int(input("Enter the starting Celsius value: "))
    end = int(input("Enter the ending Celsius value: "))

    if start > end:
        print("Starting value must be less than or equal to the ending value.")
    else:
        for celsius in range(start, end + 1):
            fahrenheit = (celsius * 9 / 5) + 32
            print(f"{celsius}°C = {fahrenheit}°F")

except ValueError:

4. Final AI-Assisted Program

RUN_PROFILE = "ai-assisted"

def celsius_to_fahrenheit(celsius: float) -> float:
    return celsius * 9 / 5 + 32


if __name__ == "__main__":
    celsius = float(input("Enter temperature in Celsius: "))
    fahrenheit = celsius_to_fahrenheit(celsius)
    print("Temperature in Fahrenheit:", fahrenheit)
